#include <Trade/Trade.mqh>
CTrade trade;

//+------------------------------------------------------------------+
//| INPUTS                                                           |
//+------------------------------------------------------------------+
input group "=== General Settings ==="
input long     InpMagic          = 123457;    
input string   InpComment        = "Mr.BP_v2.9";
input int      InpSlippage       = 50;        

input group "=== Line Object Names ==="
input string   InpResistanceName = "RT";      // ชื่อเส้นแนวต้านด้านบน (Resistance)
input string   InpSupportName    = "SP";      // ชื่อเส้นแนวรับด้านล่าง (Support)

input group "=== Money Management ==="
input double   InpFixedLot       = 0.01;      // ล็อตเริ่มต้น 0.01

input group "=== Risk Management & Profit Target ==="
input int      InpStopLossPts    = 1000;      // ระยะ Stop Loss เริ่มต้น
input double   InpAutoCloseUSD   = 0.20;      // 🎯 ปิดไม้อัตโนมัติทันทีเมื่อกำไรถึง $0.20 USD
input double   InpMinProfitUSD   = 0.50;      // กำไรถึง $0.50 ค่อยเริ่มขยับ Trailing SL
input int      InpTrailBufferPts = 300;       // ระยะห่างของ SL จากกราฟปัจจุบัน

//+------------------------------------------------------------------+
//| GLOBALS                                                          |
//+------------------------------------------------------------------+
datetime g_lastBar        = 0;
string   g_blockReason    = "";
string   g_errorStatus    = "";
double   g_resPrice       = 0;
double   g_supPrice       = 0;

//+------------------------------------------------------------------+
//| UTILITIES                                                        |
//+------------------------------------------------------------------+
bool SelectPosition(ulong &ticket) {
   for(int i = PositionsTotal() - 1; i >= 0; i--) {
      ulong t = PositionGetTicket(i);
      if(t == 0) continue;
      if(PositionGetInteger(POSITION_MAGIC) != InpMagic) continue;
      if(PositionGetString(POSITION_SYMBOL) != _Symbol) continue;
      ticket = t; return true;
   }
   return false;
}

bool GetAndValidateLines() {
   g_resPrice = 0;
   g_supPrice = 0;
   g_errorStatus = "";

   int total = ObjectsTotal(0, -1, OBJ_HLINE);
   for(int i = 0; i < total; i++) {
      string name = ObjectName(0, i, -1, OBJ_HLINE);
      double price = ObjectGetDouble(0, name, OBJPROP_PRICE);

      if(name == InpResistanceName) g_resPrice = price;
      if(name == InpSupportName)    g_supPrice = price;
   }

   if(g_resPrice == 0 || g_supPrice == 0) {
      g_errorStatus = "ERROR: Missing lines ('" + InpResistanceName + "' or '" + InpSupportName + "')";
      return false;
   }

   if(g_resPrice <= g_supPrice) {
      g_errorStatus = "ERROR: RT must be higher than SP!";
      return false;
   }

   return true;
}

//+------------------------------------------------------------------+
//| SMART POSITION MANAGEMENT (AUTO CLOSE & TRAILING SL)             |
//+------------------------------------------------------------------+
void ManagePosition() {
   ulong ticket;
   if(!SelectPosition(ticket)) return;

   long     type         = PositionGetInteger(POSITION_TYPE);
   double   profitUSD    = PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP);
   double   sl           = PositionGetDouble(POSITION_SL);
   bool     isBuy        = (type == POSITION_TYPE_BUY);

   // 🎯 ระบบปิดไม้อัตโนมัติทันทีเมื่อกำไรถึงเป้าหมาย ($0.20 ขึ้นไป)
   if(profitUSD >= InpAutoCloseUSD) {
      if(trade.PositionClose(ticket)) {
         Print("🎯 Mr.BP Auto Closed Position! Target Profit Reached: $", profitUSD);
         return;
      }
   }

   // 🔒 ระบบ Trailing SL เมื่อกำไรถึงเกณฑ์ที่กำหนด ($0.50 ขึ้นไป)
   if(profitUSD >= InpMinProfitUSD) {
      if(isBuy) {
         double px = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         double candidateSL = px - (InpTrailBufferPts * _Point);
         candidateSL = NormalizeDouble(candidateSL, _Digits);
         
         if(candidateSL > sl && candidateSL < px) {
            trade.PositionModify(ticket, candidateSL, 0);
            Print("🔒 Mr.BP Trailing SL Updated (Buy) | Profit: $", profitUSD);
         }
      }
      else {
         double px = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
         double candidateSL = px + (InpTrailBufferPts * _Point);
         candidateSL = NormalizeDouble(candidateSL, _Digits);
         
         if((candidateSL < sl || sl == 0) && candidateSL > px) {
            trade.PositionModify(ticket, candidateSL, 0);
            Print("🔒 Mr.BP Trailing SL Updated (Sell) | Profit: $", profitUSD);
         }
      }
   }
}

//+------------------------------------------------------------------+
//| EXPERT INITIALIZATION                                            |
//+------------------------------------------------------------------+
int OnInit() {
   trade.SetExpertMagicNumber((ulong)InpMagic);
   trade.SetDeviationInPoints(InpSlippage);
   Print("🚀 Mr.BP v2.9 Breakpoint EA Initialized (Auto Close at $0.20 USD Active)!");
   return INIT_SUCCEEDED;
}

void OnDeinit(const int reason) {
   Comment("");
}

//+------------------------------------------------------------------+
//| EXPERT TICK / NEW BAR LOGIC                                      |
//+------------------------------------------------------------------+
void OnNewBar() {
   g_blockReason = "";

   ulong t;
   if(SelectPosition(t)) {
      g_blockReason = "In position (Waiting breakout result)";
      return;
   }

   if(!GetAndValidateLines()) {
      g_blockReason = g_errorStatus;
      return;
   }

   MqlRates r[];
   ArraySetAsSeries(r, true);
   if(CopyRates(_Symbol, _Period, 0, 3, r) < 3) return;

   bool breakoutBuy  = (r[1].close > g_resPrice && r[2].close <= g_resPrice);
   bool breakoutSell = (r[1].close < g_supPrice && r[2].close >= g_supPrice);

   double lot = InpFixedLot;

   if(breakoutBuy) {
      double price = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      double sl    = NormalizeDouble(price - InpStopLossPts * _Point, _Digits);
      double tp    = 0; 
      
      if(trade.Buy(lot, _Symbol, 0, sl, tp, InpComment)) {
         Print("🎯 Mr.BP v2.9 BUY Breakout! Price crossed RT: ", g_resPrice);
      }
   }
   else if(breakoutSell) {
      double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      double sl    = NormalizeDouble(price + InpStopLossPts * _Point, _Digits);
      double tp    = 0; 
      
      if(trade.Sell(lot, _Symbol, 0, sl, tp, InpComment)) {
         Print("🎯 Mr.BP v2.9 SELL Breakout! Price crossed SP: ", g_supPrice);
      }
   }
}

void OnTick() {
   ManagePosition();
   GetAndValidateLines();

   string statusDisplay = (g_errorStatus != "") ? "❌ " + g_errorStatus : (g_blockReason == "" ? "🟢 ACTIVE & MONITORING" : "⏳ " + g_blockReason);

   Comment("🤖 MR.BP v2.9 [BREAKPOINT SEMI-AUTO]\n",
           "----------------------------------------\n",
           "Status: ", statusDisplay, "\n",
           "RT (Resistance): ", (g_resPrice > 0 ? DoubleToString(g_resPrice, _Digits) : "Not Found"), "\n",
           "SP (Support):    ", (g_supPrice > 0 ? DoubleToString(g_supPrice, _Digits) : "Not Found"), "\n",
           "----------------------------------------\n",
           "Lot: ", InpFixedLot, " | Auto Close Target: $", DoubleToString(InpAutoCloseUSD, 2), "\n",
           "Trailing SL Trigger: $", DoubleToString(InpMinProfitUSD, 2));

   datetime bt = iTime(_Symbol, _Period, 0);
   if(bt == g_lastBar) return;
   g_lastBar = bt;
   OnNewBar();
}
