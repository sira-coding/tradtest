const express = require('express'); const cors = require('cors'); const app = express(); app.use(express.json()); app.use(cors()); let isRunning = false; app.post('/control', (req, res) => { const { action } = req.body; if (action === 'START') isRunning = true; if (action === 'STOP') isRunning = false; console.log(`[Command] Bot status: ${isRunning ? 'RUNNING 🟢' : 'STOPPED 🔴'}`); res.json({ success: true, isRunning }); }); app.get('/status', (req, res) => { res.json({ isRunning }); }); app.listen(3000, () => { console.log('✅ Local Server active on port 3000'); });
